      \end{block}

          \begin{itemize}
            \item second level is an en-dash
              \begin{itemize}
                \item third is a small triangle
              \end{itemize}
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Also from the theme}
        The title bar above, the footer below, and the divider slide before
        this one, which every \texttt{\textbackslash section} produces.
      \end{block}
      \begin{itemize}
        \item \texttt{fitemize} and \texttt{fenumerate} spread their items
              over the whole frame
        \item \texttt{\textbackslash cornerlogo} puts a logo on every slide
        \item \texttt{\textbackslash closingslide} ends the deck — last slide
              here
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
