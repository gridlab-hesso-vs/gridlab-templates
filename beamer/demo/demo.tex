\begin{frame}{What this deck is}
  \begin{itemize}
    \item Every slide here shows a feature of \texttt{beamerthemegridlab.sty}
          and names the macro behind it.
    \item The title page you just saw, this title bar, the footer and the
          section dividers come from the theme — no setup in the
          \texttt{.tex} file.

          GridLab blue and orange and both are one line to change.

\section{Slide furniture}
% ==========================================================================

\begin{frame}{Blocks, columns, lists}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Blocks}
        Coloured title bar, light background. Group a definition, a result,
        an assumption.
      \end{block}
      \begin{itemize}
        \item Bullets are accent-coloured
          \begin{itemize}
            \item second level is an en-dash
              \begin{itemize}
                \item third is a small triangle
              \end{itemize}
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Footer}
        Short author, short title and \texttt{frame / total} sit at the
        bottom of every slide. Set them with
        \texttt{\textbackslash renewcommand}\\
        \texttt{\ \ \{\textbackslash insertshortauthor\}\{F. Sossan\}}.
      \end{block}
      \begin{block}{Corner logo}
        \texttt{\textbackslash def\textbackslash cornerlogo\{logo.png\}}
        puts a logo bottom-right of \emph{every} slide.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Filled lists: \texttt{fitemize}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\footnotesize A normal \texttt{itemize}:}
      \begin{itemize}
        \item First point
        \item Second point
        \item Third point
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\footnotesize \texttt{fitemize} spreads the items over the whole
       frame — handy when a slide holds three lines and looks empty:}
      \begin{fitemize}
        \item First point
        \item Second point
        \item Third point
      \end{fitemize}
    \end{column}
  \end{columns}
  \vfill
  {\footnotesize \texttt{fenumerate} does the same for numbered lists.}
\end{frame}

% A full-colour divider you place by hand, anywhere:
\topicslide{\texttt{\textbackslash topicslide} — a divider on demand}

\begin{frame}{Section dividers come for free}
  \begin{itemize}
    \item Every \texttt{\textbackslash section\{...\}} automatically produces
          the full-colour divider you saw before this slide.
    \item \texttt{\textbackslash topicslide\{Title\}} produces one wherever
          you want, without opening a section.
    \item \texttt{\textbackslash titleslide\{Title\}} produces an extra title
          page mid-deck.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Maths helpers}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      Upright constants: $\const{SOC}_{\max}$

      \smallskip
      A labelled underbrace, no wide gap:
      \[ \uunderbrace{P_t + Q_t}{apparent power} \]

      A forecast: \emph{value at $t$, known at $k$}:
      $\vvector{P}{t}{k}$

      \smallskip
      Phasors: $230\phase{30^\circ}$ \qquad
      Cancelling: $\cancel{x}$
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
\begin{verbatim}
$\const{SOC}_{\max}$
\[ \uunderbrace{P_t + Q_t}
     {apparent power} \]
$\vvector{P}{t}{k}$
$230\phase{30^\circ}$
$\cancel{x}$

\section{Plots and diagrams}
% ==========================================================================

\begin{frame}[fragile]{First-quadrant plot: \texttt{q1simpleplot}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \centering
      \begin{q1simpleplot}{0.95\textwidth}{0.5\textheight}{$x$}{$y$}
        \addplot[domain=0:1, samples=40, thick, color=primary] {x^2};
        \addplanelabel{0.62,0.5}{\scriptsize $y=x^2$}{anchor=west}
      \end{q1simpleplot}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
\begin{verbatim}
\begin{q1simpleplot}
    {0.95\textwidth}{0.5\textheight}
    {$x$}{$y$}
  \addplot[domain=0:1, samples=40,
           thick, color=primary]{x^2};
  \addplanelabel{0.62,0.5}
    {\scriptsize $y=x^2$}{anchor=west}
\end{q1simpleplot}
\end{verbatim}
      \normalsize
      Axes through the origin, labels at the arrow tips, no ticks to read.
      \texttt{q1q2simpleplot} does the same over $x\in[-1,1]$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{General plot and vectors}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \centering
      \begin{plot}{0.95\textwidth}{0.5\textheight}{-2}{2}{-2}{2}{1}{$P$}{$Q$}
        \vectorarrow{0,0}{1.4,1.2}{$\bar S$}
      \end{plot}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
\begin{verbatim}
\begin{plot}{w}{h}{xmin}{xmax}
            {ymin}{ymax}{tickstep}
            {$P$}{$Q$}
  \vectorarrow{0,0}{1.4,1.2}{$\bar S$}
\end{plot}
\end{verbatim}
      \normalsize
      \texttt{\textbackslash vectorarrow} draws an arrow between two axis
      coordinates and puts a white-backed label at its midpoint.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Flowchart styles}

\section{Making it yours}
% ==========================================================================

\begin{frame}[fragile]{Rebranding is two lines}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \footnotesize
\begin{verbatim}
\def\themeprimary{376092}    % main
\def\themesecondary{ED7D31}  % accent
\usetheme{gridlab}
\end{verbatim}
      \normalsize
      Set them \emph{before} the theme is loaded and every title bar,
      divider, bullet and block follows.
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
\begin{verbatim}
\def\titlelogos{...}
\def\titlebottomlogo{logo.png}
\def\cornerlogo{logo.png}
\end{verbatim}
      \normalsize
      Logos default to the ones shipped with the theme; override them, or
      set one empty to switch it off.
    \end{column}
  \end{columns}
\end{frame}
